class Action(Enum):
    ALLOW = "allow"
    BLOCK = "block"
    DECAY = "decay"

@dataclass
class Transaction:
    """
    Represents a single financial movement.
    Includes explicit Decimal handling to prevent standard Float
    comparison bugs common in rule engines.
    """
    id: str
    amount: decimal.Decimal
    timestamp: datetime
    merchant: str
    currency: str = "USD"
    category: str = "ecommerce"

    def __post_init__(self):
        # Ensure amount is truly a Decimal, fixing potential float-string drift
        if isinstance(self.amount, float):
            self.amount = decimal.Decimal(str(self.amount))
        if isinstance(self.merchant, str):

            self.merchant = self.merchant.strip()

        return {
            "id": self.id,
            "amount": str(self.amount),
            "timestamp": self.timestamp.isoformat(),
            "merchant": self.merchant
        }

class Rule:
    """
    A rule that acts on a Transaction.
    Handles the 'Rule Priority Conflict' logic.
    """
    def __init__(self, name: str, matcher: Callable, action: Action, priority: int):
        self.name = name
        self.matcher = matcher
        self.action = action
        self.priority = priority
        self.triggered_count = 0

    def matches(self, tx: Transaction) -> bool:
        # Returns True if matcher returns True
        is_match = self.matcher(tx)
        if is_match:
            self.triggered_count += 1
        return is_match

    def get_action(self, tx: Transaction) -> Optional[Action]:
        if self.matcher(tx):
            return self.action
        return None

class AgentShieldEngine:
    """
    The core evaluation logic.
    Handles:
    1. False Negatives (Did it match?)
    2. Velocity Race Conditions (Did the counter reset?)
    3. Session Budget (Did decay work?)
    """
    
    def __init__(self):
        self.rules: List[Rule] = []
        self.velocity_buffer: dict = {} # Stores last 5 timestamps per merchant
        self.session_budget: decimal.Decimal = decimal.Decimal("100.00")

    def add_rule(self, rule: Rule):
        # Ensures priority conflict resolution: Lower number = Higher precedence
        self.rules.append(rule)
        # Re-sort if priorities were added dynamically
        self.rules.sort(key=lambda r: r.priority)

    def add_velocity_rule(self, threshold: int = 5, window: int = 60):
        """

        def matcher(tx):

            # In a real engine, this checks if the count > threshold

        return Rule(
            name="velocity_proxy",

            action=Action.BLOCK,
            priority=10 # Low priority, acts as a "catch all"
        )

    def evaluate(self, tx: Transaction) -> Action:
        """
        Runs the core transaction through the rules.
        Returns the decisive Action.
        """
        # 1. Check if any rules match
        matches = [r for r in self.rules if r.matcher(tx)]

        if not matches:
            # If no rules defined, assume ALLOW (Default False Positive if rule was meant to catch it)
            return Action.ALLOW

        # 2. Priority Conflict Resolution
        # Usually, the rule with the highest priority (or lowest index depending on sorting) wins.
        # Let's use the 'priority' integer attribute.
        winning_rule = max(matches, key=lambda r: r.priority) 
        
        action = winning_rule.action
        
        # 3. Session Budget Miscalculation (Decay)

        if action == Action.DECAY:
            tx.amount = tx.amount * decimal.Decimal("0.9") # Example decay logic
            action = Action.ALLOW # Now allow the reduced amount

        return action
